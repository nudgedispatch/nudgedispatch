const $ = (sel, root = document) => root.querySelector(sel);
const esc = (v) => String(v == null ? "" : v).replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));

async function api(path, { method = "GET", body } = {}) {
  const headers = { "x-nudge-client": "1" };
  if (body !== undefined) headers["content-type"] = "application/json";
  const res = await fetch(path, { method, headers, credentials: "same-origin", body: body === undefined ? undefined : JSON.stringify(body) });
  const text = await res.text();
  let data;
  try { data = text ? JSON.parse(text) : {}; } catch { data = { ok: false, error: text || "Request failed." }; }
  return { status: res.status, data };
}

function sessionIdFromUrl() {
  const id = new URLSearchParams(location.search).get("session_id") || "";
  return /^cs_(test|live)_[A-Za-z0-9]{10,200}$/.test(id) ? id : "";
}

function isCompletePage() {
  return location.pathname === "/signup/complete" || location.pathname === "/signup/complete/";
}

async function runComplete() {
  const box = $("#signup-main");
  const sessionId = sessionIdFromUrl();
  if (!sessionId) {
    box.innerHTML = `<section class="card"><h1>Payment not finished</h1><p class="note">We couldn't find a completed payment. If you were charged, email <a href="mailto:nudgedispatch@gmail.com">nudgedispatch@gmail.com</a>.</p><p><a class="btn full linkbtn-block" href="/signup">Back to sign up</a></p></section>`;
    return;
  }
  box.innerHTML = `<section class="card"><h1>Setting up your account</h1><p class="note">This only takes a moment.</p><p class="muted" id="signup-status">Saving your payment and creating your booking page…</p></section>`;
  const result = await api("/api/signup/complete", { method: "POST", body: { sessionId } });
  if (!result.data.ok) {
    $("#signup-status").textContent = result.data.error || "Could not finish setting up your account.";
    const p = document.createElement("p");
    p.innerHTML = `<a class="btn full linkbtn-block" href="/signup">Back to sign up</a>`;
    box.querySelector(".card").appendChild(p);
    return;
  }
  location.replace("/#/welcome");
}

const LOGO_MAX_EDGE = 384;
const LOGO_MAX_DATA_URL = 180000;

function imageFromUrl(url) {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error("decode"));
    img.src = url;
  });
}
async function loadImageFromFile(file) {
  if (typeof createImageBitmap === "function") {
    try { return await createImageBitmap(file); } catch { /* fall through */ }
  }
  let url = "";
  try {
    url = URL.createObjectURL(file);
    const img = await imageFromUrl(url);
    URL.revokeObjectURL(url);
    return img;
  } catch {
    if (url) try { URL.revokeObjectURL(url); } catch { /* ignore */ }
  }
  const dataUrl = await new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(String(reader.result || ""));
    reader.onerror = () => reject(new Error("decode"));
    reader.readAsDataURL(file);
  });
  return imageFromUrl(dataUrl);
}

async function compressLogoFile(file) {
  if (file.size > 15 * 1024 * 1024) throw new Error("too-large");
  const img = await loadImageFromFile(file);
  const canvas = document.createElement("canvas");
  const ctx = canvas.getContext("2d");
  if (!ctx) throw new Error("canvas");
  let edge = LOGO_MAX_EDGE;
  let dataUrl = "";
  for (let i = 0; i < 6; i++) {
    const w = img.naturalWidth || img.width;
    const h = img.naturalHeight || img.height;
    const scale = Math.min(1, edge / Math.max(w, h, 1));
    canvas.width = Math.max(1, Math.round(w * scale));
    canvas.height = Math.max(1, Math.round(h * scale));
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
    dataUrl = canvas.toDataURL("image/png");
    if (dataUrl.length <= LOGO_MAX_DATA_URL) {
      if (img && typeof img.close === "function") try { img.close(); } catch { /* ignore */ }
      return dataUrl;
    }
    edge = Math.floor(edge * 0.75);
  }
  if (img && typeof img.close === "function") try { img.close(); } catch { /* ignore */ }
  throw new Error("too-large");
}

function hexLuminance(hex) {
  const n = parseInt(String(hex).replace("#", ""), 16);
  if (!Number.isFinite(n)) return 0;
  const ch = (c) => { const s = c / 255; return s <= 0.04045 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4); };
  return 0.2126 * ch((n >> 16) & 255) + 0.7152 * ch((n >> 8) & 255) + 0.0722 * ch(n & 255);
}
function contrastText(bg) {
  const L = hexLuminance(bg);
  const white = 1.05 / (L + 0.05);
  const black = (L + 0.05) / 0.05;
  return white >= black ? "#f5f5f5" : "#111111";
}

function turnstileToken() {
  const el = $("[name=cf-turnstile-response]");
  if (el && el.value) return el.value;
  if (window.turnstile && typeof window.turnstile.getResponse === "function") {
    try { return window.turnstile.getResponse() || ""; } catch { return ""; }
  }
  return "";
}

let slugTimer = 0;
let lastSlugChecked = "";

function paintSlugStatus(text, ok) {
  const el = $("#slug-status");
  if (!el) return;
  el.textContent = text;
  el.className = "sub slug-status" + (ok === true ? " ok" : ok === false ? " err" : "");
}

function paintSlugPreview(host) {
  const el = $("#slug-preview");
  if (!el) return;
  const slug = ($("#bookingSlug") && $("#bookingSlug").value.trim().toLowerCase()) || "";
  const base = "https://" + (host || "app.nudgedispatch.com") + "/book/";
  el.textContent = slug ? ("Booking link: " + base + slug + "/") : ("Booking link: " + base + "…/");
}

async function checkSlug(slug, source) {
  const value = String(slug || "").trim().toLowerCase();
  if (!value || value === lastSlugChecked) return;
  lastSlugChecked = value;
  paintSlugStatus("Checking availability…");
  const result = await api("/api/signup/slug?slug=" + encodeURIComponent(value));
  if ($("#bookingSlug") && $("#bookingSlug").value.trim().toLowerCase() !== value) return;
  if (!result.data.ok) {
    paintSlugStatus(result.data.error || "Could not check that company code.", false);
    return;
  }
  if (result.data.available) {
    paintSlugStatus("This company code is available. Your booking page will be " + (result.data.path || "/book/" + value + "/"), true);
  } else {
    paintSlugStatus(result.data.error || "That company code is taken.", false);
  }
  if (source === "suggest" && result.data.suggestion && result.data.suggestion !== value && !result.data.available) {
    $("#bookingSlug").value = result.data.suggestion;
    lastSlugChecked = "";
    checkSlug(result.data.suggestion, "follow");
  }
}

function scheduleSlugCheck() {
  clearTimeout(slugTimer);
  slugTimer = setTimeout(() => checkSlug($("#bookingSlug").value), 300);
}

async function boot() {
  if (isCompletePage()) return runComplete();
  const result = await api("/api/signup/config");
  if (!result.data.ok) {
    $("#signup-main").innerHTML = `<section class="card"><h1>Sign up</h1><p class="err">${esc(result.data.error || "Could not load sign up.")}</p></section>`;
    return;
  }
  renderForm(result.data);
}
