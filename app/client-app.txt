var NudgeA2hs = (function () {
  "use strict";
  var STORAGE_KEY = "nudge-a2hs-dismissed-at";
  var REOFFER_MS = 14 * 24 * 60 * 60 * 1000;
  var IOS_OTHER = /CriOS|FxiOS|EdgiOS|OPiOS|OPT\/|DuckDuckGo|YaBrowser|GSA\/|FBAN|FBAV|FB_IAB|Instagram|Line\/|Twitter|Snapchat|LinkedInApp|musical_ly|ByteDance|TikTok|WebView/;

  function isIos(ua, platform, maxTouchPoints) {
    ua = String(ua || "");
    platform = String(platform || "");
    if (/iPad|iPhone|iPod/.test(ua)) return true;
    if (platform === "MacIntel" && Number(maxTouchPoints) > 1) return true;
    return false;
  }
  function isIosSafari(ua, platform, maxTouchPoints) {
    if (!isIos(ua, platform, maxTouchPoints)) return false;
    ua = String(ua || "");
    if (IOS_OTHER.test(ua)) return false;
    return /Safari/.test(ua);
  }
  function isDesktop(ua, platform, maxTouchPoints) {
    if (isIos(ua, platform, maxTouchPoints)) return false;
    if (/Android/i.test(String(ua || ""))) return false;
    return true;
  }
  function isStandalone(matchesStandalone, navigatorStandalone) {
    return !!(matchesStandalone || navigatorStandalone === true);
  }
  function readDismissedAt(storage) {
    try {
      var raw = storage && storage.getItem(STORAGE_KEY);
      var n = Number(raw);
      return Number.isFinite(n) && n > 0 ? n : 0;
    } catch (err) {
      return 0;
    }
  }
  function writeDismissedAt(storage, now) {
    try { if (storage) storage.setItem(STORAGE_KEY, String(now)); } catch (err) { /* quota / private mode */ }
  }
  function shouldShow(opts) {
    opts = opts || {};
    var now = Number(opts.now) || 0;
    if (opts.standalone) return { show: false, variant: "", reason: "standalone" };
    if (opts.desktop) return { show: false, variant: "", reason: "desktop" };
    var dismissedAt = Number(opts.dismissedAt) || 0;
    if (dismissedAt && (now - dismissedAt) < REOFFER_MS) return { show: false, variant: "", reason: "dismissed" };
    if (opts.ios) {
      return opts.iosSafari
        ? { show: true, variant: "ios-safari", reason: "" }
        : { show: true, variant: "ios-other", reason: "" };
    }
    if (opts.hasNativePrompt) return { show: true, variant: "android", reason: "" };
    return { show: false, variant: "", reason: "no-prompt" };
  }
  return { STORAGE_KEY: STORAGE_KEY, REOFFER_MS: REOFFER_MS, isIos: isIos, isIosSafari: isIosSafari, isDesktop: isDesktop, isStandalone: isStandalone, readDismissedAt: readDismissedAt, writeDismissedAt: writeDismissedAt, shouldShow: shouldShow };
})();

function a2hsEnv() {
  const ua = navigator.userAgent || "";
  const platform = navigator.platform || "";
  const maxTouchPoints = navigator.maxTouchPoints || 0;
  const standaloneMatch = window.matchMedia && window.matchMedia("(display-mode: standalone)").matches;
  return {
    standalone: NudgeA2hs.isStandalone(standaloneMatch, window.navigator.standalone === true),
    desktop: NudgeA2hs.isDesktop(ua, platform, maxTouchPoints),
    dismissedAt: NudgeA2hs.readDismissedAt(window.localStorage),
    now: Date.now(),
    ios: NudgeA2hs.isIos(ua, platform, maxTouchPoints),
    iosSafari: NudgeA2hs.isIosSafari(ua, platform, maxTouchPoints),
    hasNativePrompt: !!(deferredInstallPrompt && typeof deferredInstallPrompt.prompt === "function"),
  };
}

function showA2hsPromptIfNeeded() {
  hideA2hsPrompt();
  if (!state.me || state.me.platformAdmin) return;
  const decision = NudgeA2hs.shouldShow(a2hsEnv());
  if (!decision.show) return;
  const overlay = document.createElement("div");
  overlay.id = "a2hs";
  overlay.className = "a2hs-overlay";
  overlay.setAttribute("role", "dialog");
  overlay.setAttribute("aria-modal", "true");
  overlay.setAttribute("aria-labelledby", "a2hs-title");
  overlay.setAttribute("data-variant", decision.variant);
  overlay.innerHTML = a2hsCardHtml(decision.variant);
  document.body.appendChild(overlay);
  const later = overlay.querySelector("#a2hs-later");
  if (later) {
    later.addEventListener("click", dismissA2hsPrompt);
    later.focus();
  }
  const install = overlay.querySelector("#a2hs-install");
  if (install) {
    install.addEventListener("click", async () => {
      if (!deferredInstallPrompt || typeof deferredInstallPrompt.prompt !== "function") return;
      deferredInstallPrompt.prompt();
      let outcome = "";
      try {
        const choice = await deferredInstallPrompt.userChoice;
        outcome = (choice && choice.outcome) || "";
      } catch { /* prompt may be dismissed without a choice */ }
      deferredInstallPrompt = null;
      if (outcome === "accepted") {
        NudgeA2hs.writeDismissedAt(window.localStorage, Date.now());
        hideA2hsPrompt();
      }
    });
    install.focus();
  }
  overlay.addEventListener("keydown", (event) => {
    if (event.key === "Escape") {
      event.preventDefault();
      dismissA2hsPrompt();
    }
  });
}

function longIssued(iso) {
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso || "");
  if (!match) return iso || "";
  const dt = new Date(Date.UTC(+match[1], +match[2] - 1, +match[3]));
  const dow = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"][dt.getUTCDay()];
  const mon = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"][+match[2] - 1];
  return `${dow}, ${mon} ${+match[3]}, ${match[1]}`;
}
function time12(value) {
  const match = /^(\d{2}):(\d{2})$/.exec(value || "");
  if (!match) return value || "";
  let hour = +match[1];
  const suffix = hour >= 12 ? "PM" : "AM";
  hour = hour % 12 || 12;
  return `${hour}:${match[2]} ${suffix}`;
}

const locWatch = { id: null, timer: null, on: false, rideId: "" };

function locStopWatch() {
  if (locWatch.id != null && navigator.geolocation) navigator.geolocation.clearWatch(locWatch.id);
  locWatch.id = null;
  if (locWatch.timer) clearInterval(locWatch.timer);
  locWatch.timer = null;
}

function locSendFix(pos) {
  if (!locWatch.rideId) return Promise.resolve();
  return api("/api/location", {
    method: "POST",
    body: {
      rideId: locWatch.rideId,
      lat: pos.coords.latitude,
      lng: pos.coords.longitude,
      acc: typeof pos.coords.accuracy === "number" ? pos.coords.accuracy : null,
      ts: pos.timestamp || Date.now(),
    },
  });
}

function locStartWatch() {
  locStopWatch();
  if (!navigator.geolocation || window.isSecureContext === false) return;
  const opt = { enableHighAccuracy: false, maximumAge: 30000, timeout: 30000 };
  locWatch.id = navigator.geolocation.watchPosition((pos) => { locSendFix(pos); }, () => {}, opt);
  locWatch.timer = setInterval(() => {
    if (!locWatch.on || document.visibilityState !== "visible" || !navigator.geolocation) return;
    navigator.geolocation.getCurrentPosition((pos) => { locSendFix(pos); }, () => {}, opt);
  }, 45000);
}

document.addEventListener("visibilitychange", () => {
  if (!locWatch.on) return;
  if (document.visibilityState !== "visible") locStopWatch();
  else locStartWatch();
});

async function api(path, options = {}) {
  const headers = { "X-Nudge-Client": "1" };
  if (options.body) headers["Content-Type"] = "application/json";
  if (options.auth !== false && state.me && options.method && options.method !== "GET") headers["X-CSRF-Token"] = state.me.csrf;
  let response;
  try {
    response = await fetch(path, {
      method: options.method || "GET",
      headers,
      body: options.body ? JSON.stringify(options.body) : undefined,
      credentials: "same-origin",
    });
  } catch {
    return { status: 0, data: { ok: false, error: "No connection. Check your signal and try again." } };
  }
  const data = await response.json().catch(() => ({ ok: false, error: "Unexpected response (" + response.status + ")." }));
  if (!response.ok && data && typeof data === "object" && data.ok !== false) { data.ok = false; data.error = data.error || "Request failed (" + response.status + ")."; }
  if (response.status === 403 && data && data.billingLocked && state.me && !state.me.platformAdmin) {
    state.me.access = { ...(state.me.access || {}), state: "locked", locked: true, banner: null, lockedView: data.lockedView || null };
    setTimeout(() => renderLocked(), 0);
  }
  if (response.status === 401 && path !== "/api/login" && path !== "/api/invites/accept" && path !== "/api/driver-login") {
    const hadSession = !!state.me;
    state.me = null;
    const screen = (location.hash || "").replace(/^#\//, "").split("/")[0];
    if (hadSession || (screen && screen !== "login" && screen !== "start")) location.hash = "#/login";
  }
  return { status: response.status, data };
}

async function loadPaymentCard(gen, note) {
  const card = document.querySelector("#card-card");
  if (!card) return;
  const params = new URLSearchParams(location.search);
  const sessionId = params.get("card_session");
  let r;
  if (sessionId) {
    history.replaceState(null, "", location.pathname + location.hash);
    r = (await api("/api/billing/payment-method/confirm", { method: "POST", body: { sessionId } })).data;
    if (r.ok) note = "Card saved.";
    else { note = r.error || "Could not save the card."; r = null; }
  }
  if (!r) r = (await api("/api/billing/payment-method")).data;
  if (gen !== undefined && gen !== renderGen) return;
  card.innerHTML = paymentCardHtml(r);
  if (note) setMsg(card.querySelector(".pm-msg") || card, note, /saved|paid/i.test(note));
}

const INVITE_TOKEN_KEY = "nudge_invite_token";

function rememberInviteToken(token) {
  state.inviteToken = token || "";
  try {
    if (token) sessionStorage.setItem(INVITE_TOKEN_KEY, token);
    else sessionStorage.removeItem(INVITE_TOKEN_KEY);
  } catch {}
}

function loadInviteToken() {
  if (state.inviteToken) return state.inviteToken;
  try { state.inviteToken = sessionStorage.getItem(INVITE_TOKEN_KEY) || ""; }
  catch { state.inviteToken = state.inviteToken || ""; }
  return state.inviteToken;
}

function clearInviteToken() {
  state.inviteToken = "";
  try { sessionStorage.removeItem(INVITE_TOKEN_KEY); } catch {}
}

function showDesktopSiteHint() {
  try {

    const smallScreen = Math.min(screen.width, screen.height) <= 600;

  const ios = alertIsIos();
  const bar = document.createElement("div");
  bar.className = "desktop-site-hint";
  bar.id = "desktop-site-hint";

async function boot() {
  const start = /^#\/start\/([A-Za-z0-9_-]+)$/.exec(location.hash);
  if (start) {
    rememberInviteToken(start[1]);
    history.replaceState(null, "", location.pathname + location.search + "#/start");
  } else if (location.hash === "#/start") {
    loadInviteToken();
  }
  if ("serviceWorker" in navigator) navigator.serviceWorker.register("/sw.js").catch(() => {});
  showDesktopSiteHint();
  const me = await api("/api/me");
  if (me.data.ok) {
    state.me = me.data;
    setTag();
    if (location.hash === "" || location.hash === "#/login" || location.hash === "#/start") {
      location.hash = me.data.platformAdmin ? "#/admin" : "#/rides";
    }
    if (location.hash === "#/welcome") offerA2hsAfterLogin();
    await route();
    return;
  }
  if (whereToken()) return route();
  if (location.hash === "#/start") return renderStart();
  location.hash = "#/login";
  renderLogin();
}

function startAcceptError(data) {
  const error = data && data.error || "";
  if (/invalid or already used/i.test(error)) {
    return "This setup link has expired or was already used. Ask your Nudge Dispatch contact for a new link.";
  }
  return error || "Could not save the password.";
}

function updateStartHint() {
  const password = document.querySelector("#password");
  const again = document.querySelector("#again");
  const hint = document.querySelector("#pw-hint");
  if (!password || !again || !hint) return;
  const a = password.value;
  const b = again.value;
  hint.style.color = "";
  hint.textContent = "";
  if (!a && !b) return;
  if (a.length < 12) {
    hint.style.color = "var(--red)";
    hint.textContent = "Use at least 12 characters.";
    return;
  }
  if (a.length > 200) {
    hint.style.color = "var(--red)";
    hint.textContent = "Use at most 200 characters.";
    return;
  }
  if (b && a !== b) {
    hint.style.color = "var(--red)";
    hint.textContent = "Those passwords do not match.";
    return;
  }
  if (b && a === b) {
    hint.style.color = "var(--green)";
    hint.textContent = "Passwords match.";
  }
}

  const againInput = document.querySelector("#again");
  const button = form.querySelector("button[type=submit]");
  passwordInput.addEventListener("input", updateStartHint);
  againInput.addEventListener("input", updateStartHint);
  if (!state.inviteToken) formError("Open the one-time setup link you were given.");
  // New accounts (owner or driver setup links) must agree to the Terms + Privacy; password-reset links don't show it.
  const termsRow = document.querySelector("#terms-row");
  const termsBox = document.querySelector("#termsAccepted");
  let needsTerms = true;
  if (state.inviteToken) {
    api("/api/invites/peek", { method: "POST", auth: false, body: { token: state.inviteToken } }).then((peek) => {
      if (peek && peek.data && peek.data.ok && peek.data.kind === "reset") {
        needsTerms = false;
        termsRow.style.display = "none";
      }
    }).catch(() => {});
  }
  form.onsubmit = async (event) => {
    event.preventDefault();
    const password = passwordInput.value;
    const again = againInput.value;
    updateStartHint();
    if (!password) return formError("Enter a password.");
    if (password.length < 12 || password.length > 200) return formError("Use a password between 12 and 200 characters.");
    if (password !== again) return formError("Those passwords do not match.");
    if (!state.inviteToken) return formError("Open the one-time setup link you were given.");
    if (needsTerms && !termsBox.checked) return formError("Please agree to the Terms of Service and Privacy Policy.");

      const result = await api("/api/invites/accept", { method: "POST", auth: false, body: acceptBody });

      state.me = result.data;
      offerA2hsAfterLogin();
      location.hash = "#/rides";
    } catch {
      formError("Could not save the password. Check your connection and try again.");
    } finally {
      button.disabled = false;
    }
  };
}

async function loadTrial() {
  state.trial = null;
  state.trialBanner = null;
  if (!state.me || state.me.role !== "owner" || state.me.platformAdmin) return null;
  const result = await api("/api/trial");
  if (result.data.ok) {
    state.trial = result.data.trial;
    state.trialBanner = result.data.banner;
  }
  return state.trial;
}
function trialClosed() {
  return !!(state.trial && state.trial.open === false);
}
function trialBannerHtml() {
  const b = state.trialBanner;
  if (!b) return "";
  const contact = b.contact ? `<p class="sub">${esc(b.contact).replace(esc(b.contactEmail), `<a href="mailto:${esc(b.contactEmail)}">${esc(b.contactEmail)}</a>`)}</p>` : "";
  if (b.state === "ended") {
    return `<section class="card trial trial-ended" id="trial-card" data-state="ended">
      <h2>${esc(b.title)}</h2>
      <p class="trial-line">${esc(b.line)}</p>
      <p class="sub">${esc(b.detail)}</p>
      ${contact}
      <p class="sub"><a href="#/settings">Export your rides and clients</a> any time from More.</p>
    </section>`;
  }
  if (b.state === "converted" || b.state === "converting") {
    // Informational only: no choices, never blocks booking.
    return `<section class="card trial trial-${esc(b.state)}" id="trial-card" data-state="${esc(b.state)}" role="status">
      ${b.dismissible ? `<button class="trial-dismiss" type="button" id="trial-dismiss" aria-label="Dismiss">×</button>` : ""}
      <p class="trial-line"><strong>${esc(b.title)}</strong></p>
      <p class="sub">${esc(b.line)}</p>
      <p class="sub">${esc(b.detail)}</p>
    </section>`;
  }
  const t = state.trial;
  const pct = t && t.rideLimit ? Math.min(100, Math.round((t.ridesUsed / t.rideLimit) * 100)) : 0;
  return `<section class="card trial trial-${esc(b.state)}" id="trial-card" data-state="${esc(b.state)}">
      <p class="trial-line">${esc(b.line)}</p>
      <div class="trial-bar" aria-hidden="true"><span class="trial-fill w${Math.round(pct / 5) * 5}"></span></div>
      <p class="sub">${esc(b.detail)}</p>
      ${b.cancelNote ? `<p class="sub trial-cancel-note">${esc(b.cancelNote)}</p>` : ""}
      ${contact}
    </section>`;
}

function rideCardHtml(ride, pending) {
  const when = `${esc(ride.date)} · ${esc(time12(ride.time))}`;
  const fare = ride.status === "requested" || !ride.fare || ride.fare === "$0.00" ? "" : " · " + esc(ride.fare);
  const phone = ride.clientPhone ? `<div class="sub">${esc(ride.clientPhone)}</div>` : "";
  const notes = pending && ride.notes ? `<div class="sub">${esc(ride.notes)}</div>` : "";
  const trip = (tripLine(ride) ? `<div class="sub">${esc(tripLine(ride))}</div>` : "") + airportHtml(ride, "div");
  const owner = !(state.me && state.me.role === "driver");
  const quote = pending && owner ? (ride.quote ? quoteBoxHtml(ride.quote) : `<div class="sub">${rideStops(ride).length ? "No estimate: this request has stops, so the client saw Price on request." : "No estimate was shown to the client."}</div>`) : "";
  const actions = pending && owner ? `${trialClosed() ? trialPausedHtml() : approveBlockHtml(ride, `data-fare="${esc(ride.id)}"`, `data-approve="${esc(ride.id)}"`)}<div class="pend-actions"><button class="btn ghost" type="button" data-decline="${esc(ride.id)}">Decline</button></div>` : "";
  return `<div class="list-btn${pending ? " pending-request card" : ""}" data-id="${esc(ride.id)}">
      <div class="row"><span class="title">${esc(ride.clientName)}</span><span class="badge ${esc(ride.status)}">${esc(rideBadgeText(ride))}</span></div>
      <div class="sub when">${when}${fare}</div>
      <div class="sub">${esc(ride.pickup)} → ${esc(ride.dropoff)}</div>
      ${stopsSummaryHtml(ride, "div")}
      ${ride.driverName ? `<div class="sub">Driver: ${esc(ride.driverName)}</div>` : ""}
      ${trip}${phone}${notes}${quote}${actions}
    </div>`;
}

      ${ride.status === "completed" && !driverView ? `<button class="btn" id="invoice" type="button">${invoice ? "Open invoice" : "Create invoice"}</button>` : ""}
    </div>
    ${!driverView && ride.status === "scheduled" ? `<form id="stops-form" class="card noprint"><h2>Stops</h2>${stopEditorHtml(rideStops(ride), "ride-stops")}<button class="btn ghost full" type="submit" id="save-stops">Save stops</button><p class="err" id="stopserr"></p></form>` : ""}
    ${fleetOwner && ride.status !== "requested" && ride.status !== "declined" ? `<form id="assign" class="card noprint"><label>Assigned driver<select id="driver"><option value="">Unassigned</option>${formerDriver}${roster.map((member) => `<option value="${esc(member.id)}">${esc(driverOptionLabel(member))}</option>`).join("")}</select></label><p class="err" id="assignerr"></p></form>` : ""}
    <section class="card noprint${ride.status === "requested" || ride.status === "declined" ? " hidden" : ""}">
      <h2>Your location</h2>
      <p class="note">Sharing stays on only while this ride screen is open and the phone is unlocked. Updates pause when you leave this page or lock the phone. Nudge does not track in the background, and it does not text the location link to anyone.</p>
      <div class="loc-actions">
        <button class="btn" id="share" type="button">${sharing ? "Stop sharing" : "Start sharing"}</button>
        <button class="btn ghost" id="makelink" type="button">Create a private view link</button>
      </div>
      <p class="sub" id="locstate">${sharing ? "Sharing is on for this ride only. A view link is not sent anywhere." : "Sharing is off until you tap Start sharing."}</p>
      <div id="links">${links.map((link) => `<p class="sub">Link ${esc(link.id.slice(0, 8))} · <button type="button" class="btn ghost" data-revoke="${esc(link.id)}">Revoke</button></p>`).join("")}</div>
      <p class="err" id="locerr"></p>
    </section>
    <form id="pay" class="card noprint${driverView || ride.status === "requested" || ride.status === "declined" ? " hidden" : ""}">
      <label>Payment status
        <select id="payment">
          <option value="unpaid">Unpaid</option>
          <option value="invoiced">Invoiced</option>
          <option value="paid">Paid</option>
        </select>
      </label>
      <button class="btn full" type="submit">Save payment status</button>
      <p class="err" id="err"></p>
    </form>`;
  wireApproveBlock(box);
  const stopsForm = document.querySelector("#stops-form");
  if (stopsForm) {
    wireStopEditor(document.querySelector("#ride-stops"));
    stopsForm.onsubmit = async (event) => {
      event.preventDefault();
      const next = await api("/api/rides/" + id, { method: "PATCH", body: { stops: readStopEditor(document.querySelector("#ride-stops")) } });
      if (gen !== renderGen) return;
      const err = document.querySelector("#stopserr");
      if (!next.data.ok) { if (err) err.textContent = next.data.error || "Could not save the stops."; return; }
      route();
    };
  }
  document.querySelector("#payment").value = ride.payment || "unpaid";
  const assign = document.querySelector("#driver");
  if (assign) {
    assign.value = ride.driverUserId || "";
    assign.onchange = async () => {
      const next = await api("/api/rides/" + id, { method: "PATCH", body: { driverUserId: assign.value || null } });
      const err = document.querySelector("#assignerr");
      if (!next.data.ok) { if (err) err.textContent = next.data.error || "Could not assign the driver."; return; }
      route();
    };
  }
  if (sharing) { locWatch.rideId = id; locWatch.on = true; locStartWatch(); }
  const step = document.querySelector("#step");
  if (step) step.onclick = async () => {
    const goingTo = schedule.nextStep;
    const maps = openMapsWhenStepSaved(goingTo);
    const next = await api("/api/rides/" + id + "/progress", { method: "POST", body: { step: goingTo, from: schedule.progress || "" } });
    if (gen !== renderGen) { maps.cancel(); return; }
    if (!next.data.ok) { maps.cancel(); return formError(next.data.error); }
    const nav = next.data.ride && next.data.ride.schedule && next.data.ride.schedule.nav;
    if (shouldAutoOpenMaps(goingTo, false)) maps.go(nav && nav.url);
    else maps.cancel();
    route();
  };
  const undo = document.querySelector("#undo");
  if (undo) undo.onclick = async () => {
    const next = await api("/api/rides/" + id + "/progress", { method: "POST", body: { step: schedule.undoStep, from: schedule.progress || "", undo: true } });
    if (gen !== renderGen) return;
    if (!next.data.ok) return formError(next.data.error);
    route();
  };
  const cancel = document.querySelector("#cancel");
  if (cancel) cancel.onclick = async () => {
    const next = await api("/api/rides/" + id + "/cancel", { method: "POST", body: {} });
    if (gen !== renderGen) return;
    if (!next.data.ok) return formError(next.data.error);
    route();
  };
  const invoiceBtn = document.querySelector("#invoice");
  if (invoiceBtn) invoiceBtn.onclick = async () => {
    if (invoice) { location.hash = "#/invoices/" + invoice.id; return; }
    const next = await api("/api/rides/" + id + "/invoice", { method: "POST", body: {} });
    if (gen !== renderGen) return;
    if (!next.data.ok) return formError(next.data.error);
    location.hash = "#/invoices/" + next.data.invoice.id;
  };
  const approveBtn = document.querySelector("#approve");
  if (approveBtn) approveBtn.onclick = async () => {
    const fareEl = document.querySelector("#final-fare");
    if (!fareEl || !fareOk(fareEl.value)) return formError("Enter the final price to approve.");
    const next = await api("/api/rides/" + id + "/approve", { method: "POST", body: { fare: fareEl.value.trim() } });
    if (gen !== renderGen) return;
    if (!next.data.ok) return formError(next.data.error);
    route();
  };
  const declineBtn = document.querySelector("#decline");
  if (declineBtn) declineBtn.onclick = async () => {
    const next = await api("/api/rides/" + id + "/decline", { method: "POST", body: {} });
    if (gen !== renderGen) return;
    if (!next.data.ok) return formError(next.data.error);
    location.hash = "#/rides";
  };
  const shareBtn = document.querySelector("#share");
  if (shareBtn) shareBtn.onclick = async () => {
    const next = await api("/api/location/sharing", { method: "POST", body: { on: !sharing, rideId: id } });
    const err = document.querySelector("#locerr");
    if (!next.data.ok) { if (err) err.textContent = next.data.error || "Could not change sharing."; return; }
    if (!sharing) { locWatch.rideId = id; locWatch.on = true; locStartWatch(); }
    else { locWatch.on = false; locStopWatch(); }
    route();
  };
  document.querySelector("#makelink").onclick = async () => {
    const next = await api("/api/location/links", { method: "POST", body: { rideId: id } });
    const err = document.querySelector("#locerr");
    if (!next.data.ok) { if (err) err.textContent = next.data.error || "Could not create a link."; return; }
    const path = next.data.link.path;
    const boxLink = document.querySelector("#links");
    if (boxLink) boxLink.insertAdjacentHTML("afterbegin", `<p>Private link, shown once. It is not sent. <input readonly value="${esc(location.origin + path)}"></p>`);
  };
  box.querySelectorAll("[data-revoke]").forEach((button) => {
    button.onclick = async () => {
      const next = await api("/api/location/links/" + button.dataset.revoke + "/revoke", { method: "POST", body: {} });
      if (!next.data.ok) return formError(next.data.error);
      route();
    };
  });
  document.querySelector("#pay").onsubmit = async (event) => {
    event.preventDefault();
    const payment = document.querySelector("#payment");
    if (!payment || gen !== renderGen) return;
    const next = await api("/api/rides/" + id, { method: "PATCH", body: { payment: payment.value } });
    if (gen !== renderGen) return;
    if (!next.data.ok) return formError(next.data.error);
    route();
  };
}

// "Send PayPal invoice" (explicit tap) or, once sent, the PayPal invoice status and payer link.
function payPalInvoiceHtml(invoice) {
  const pp = invoice.paypal || {};
  if (pp.sent) {
    return `<div class="card noprint" id="pp-invoice"><p><strong>PayPal invoice</strong> · <span id="pp-status">${esc(pp.statusLabel || pp.status)}</span>${pp.refunded ? ` · <span class="err">Refunded</span>` : ""}</p>
      <p class="sub">${esc(pp.invoiceId)}${pp.mode === "sandbox" ? " · sandbox" : ""}${pp.mock ? " · simulated on this preview" : ""}. The plain PayPal link is hidden on this invoice; the rider pays the PayPal invoice.</p>
      <div class="actions">${pp.payerUrl ? `<a class="btn ghost" id="pp-open" href="${esc(pp.payerUrl)}" target="_blank" rel="noopener noreferrer">Open PayPal invoice</a>` : ""}<button class="btn ghost" id="pp-sync" type="button">Check PayPal status</button></div>
      <p class="err" id="pp-send-msg"></p></div>`;
  }
  if (!pp.available) return "";
  const email = String((invoice.billTo && invoice.billTo.email) || "").trim();
  const paid = invoice.payment === "paid";
  const hasEmail = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
  return `<div class="card noprint" id="pp-invoice">
      <button class="btn full" id="pp-send" type="button"${paid ? " disabled" : ""}>Send PayPal invoice</button>
      <p class="sub">${paid ? "This ride is already marked paid." : `PayPal emails the invoice to ${hasEmail ? esc(email) : "the rider's email on this client (needed)"}. The rider can pay by card or PayPal and add a tip. It's marked paid here when PayPal says it's fully paid.`}${pp.mock ? " Preview: simulated, nothing reaches PayPal." : ""}${pp.status === "DRAFT" ? " A PayPal draft exists; tapping again sends that same invoice." : ""}</p>
      <p class="err" id="pp-send-msg"></p></div>`;
}

// mailto:/sms: links that open the owner's own mail or messages app. Nothing is sent by the server.
function invoiceSendLinks(invoice) {
  const business = invoice.business || {};
  const name = String(business.name || "").replace(/\.+$/, "");
  const lines = Array.isArray(invoice.payLines) ? invoice.payLines : [];
  const email = String((invoice.billTo && invoice.billTo.email) || "").trim();
  const phone = String((invoice.billTo && invoice.billTo.phone) || "").replace(/[^\d+]/g, "");
  let body = String(invoice.snapshotText || "").split("\n").filter((line) => !/^Nothing was sent automatically/.test(line)).join("\n").trim();
  if (lines.length && !/(^|\n)Pay with\n/.test(body)) body += "\n\nPay with\n" + lines.join("\n");
  const mailto = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
    ? "mailto:" + encodeURIComponent(email).replace(/%40/g, "@") + "?subject=" + encodeURIComponent(`Invoice ${invoice.number} from ${name}`) + "&body=" + encodeURIComponent(body.replace(/\r?\n/g, "\r\n"))
    : "";
  const text = [`${name}: invoice ${invoice.number} for ${invoice.fare}.`];
  if (lines.length) text.push("Pay with:", ...lines);
  else if (business.paymentInstructions) text.push("How to pay: " + business.paymentInstructions);
  text.push("Thank you!");
  const sms = phone.replace(/\D/g, "").length >= 7 ? "sms:" + phone + "?&body=" + encodeURIComponent(text.join("\n")) : "";
  return { mailto, sms };
}

function download(name, contents, type) {
  const blob = new Blob([contents], { type: type + ";charset=utf-8" });
  const link = document.createElement("a");
  link.href = URL.createObjectURL(blob);
  link.download = name;
  document.body.appendChild(link);
  link.click();
  link.remove();
  setTimeout(() => URL.revokeObjectURL(link.href), 30000);
}

function paintDrivers(drivers, flash) {
  const list = document.querySelector("#driver-list");
  if (!list) return;
  if (!drivers) {
    list.innerHTML = `<p class="sub">Could not load drivers.</p>`;
    return;
  }
  const people = drivers.filter((member) => member.role === "driver");
  list.classList.toggle("muted", people.length === 0);
  list.innerHTML = people.length
    ? people.map((member) => driverCard(member, flash)).join("")
    : `<p class="sub">No drivers yet. Add one above.</p>`;
  for (const form of list.querySelectorAll(".edit-driver")) {
    form.onsubmit = async (event) => {
      event.preventDefault();
      const id = form.dataset.id;
      const err = form.querySelector(".err");
      const next = await api("/api/drivers/" + id, {
        method: "PATCH",
        body: {
          name: form.querySelector("[name=name]").value,
          vehicle: form.querySelector("[name=vehicle]").value,
          plate: form.querySelector("[name=plate]").value,
        },
      });
      if (!next.data.ok) {
        if (err) { err.classList.remove("ok"); err.textContent = next.data.error || "Could not save the driver."; }
        return;
      }
      await refreshDrivers({ id, kind: "saved" });
    };
  }
  for (const form of list.querySelectorAll(".passcode-driver")) {
    form.onsubmit = async (event) => {
      event.preventDefault();
      const id = form.dataset.id;
      const err = form.querySelector(".err");
      const next = await api("/api/drivers/" + id + "/passcode", {
        method: "PUT",
        body: { passcode: form.querySelector("[name=passcode]").value },
      });
      if (!next.data.ok) {
        if (err) { err.classList.remove("ok"); err.textContent = next.data.error || "Could not save the passcode."; }
        return;
      }
      const typed = form.querySelector("[name=passcode]").value;
      await refreshDrivers({ id, kind: "passcode", message: "Passcode saved. Tell the driver in person. It is not texted." });
      const input = document.querySelector(`[data-passcode-for="${CSS.escape(id)}"]`);
      if (input) input.value = typed;
    };
    const clear = form.querySelector("[data-clear]");
    if (clear) clear.onclick = async () => {
      const id = form.dataset.id;
      const err = form.querySelector(".err");
      const next = await api("/api/drivers/" + id + "/passcode", { method: "DELETE" });
      if (!next.data.ok) {
        if (err) { err.classList.remove("ok"); err.textContent = next.data.error || "Could not remove the passcode."; }
        return;
      }
      await refreshDrivers({ id, kind: "passcode", message: "Passcode removed. This driver cannot sign in until you set a new one." });
    };
  }
  for (const row of list.querySelectorAll(".driver-row")) {
    const remove = row.querySelector("[data-remove]");
    const confirm = row.querySelector("[data-confirm]");
    const cancel = row.querySelector("[data-cancel-delete]");
    const yes = row.querySelector("[data-confirm-delete]");
    if (remove && confirm) remove.onclick = () => {
      remove.classList.add("hidden");
      confirm.classList.remove("hidden");
    };
    if (cancel && confirm && remove) cancel.onclick = () => {
      confirm.classList.add("hidden");
      remove.classList.remove("hidden");
    };
    if (yes) yes.onclick = async () => {
      const id = row.dataset.id;
      const err = confirm.querySelector(".err");
      yes.disabled = true;
      const next = await api("/api/drivers/" + id, { method: "DELETE" });
      if (!next.data.ok) {
        yes.disabled = false;
        if (err) err.textContent = next.data.error || "Could not remove the driver.";
        return;
      }
      const n = Number(next.data.unassignedRides) || 0;
      const note = document.querySelector("#driver-err");
      if (note) {
        note.classList.add("ok");
        note.textContent = n === 0
          ? "Driver removed. They cannot sign in. Completed rides still show their name."
          : "Driver removed. " + n + " upcoming ride" + (n === 1 ? " is" : "s are") + " unassigned.";
      }
      await refreshDrivers();
    };
  }
  if (flash && flash.focusId) {
    const input = list.querySelector(`[data-passcode-for="${CSS.escape(flash.focusId)}"]`);
    const row = input && input.closest(".driver-row");
    if (row) row.scrollIntoView({ block: "nearest" });
    if (input) input.focus();
  }
}

async function refreshDrivers(flash) {
  const listed = await api("/api/drivers");
  const list = document.querySelector("#driver-list");
  if (!list) return listed;
  if (!listed.data.ok) {
    list.innerHTML = `<p class="sub">${esc(listed.data.error || "Could not load drivers.")}</p>`;
    return listed;
  }
  paintDrivers(listed.data.drivers || [], flash);
  return listed;
}

function readPaymentMethods() {
  const out = {};
  for (const f of PAY_FIELDS) {
    const on = document.querySelector(`[data-pay-on="${f.kind}"]`);
    const handle = document.querySelector(`[data-pay-handle="${f.kind}"]`);
    out[f.kind] = { on: !!(on && on.checked), handle: handle ? handle.value : "" };
  }
  return out;
}

    loadPayPalCard(gen, "PayPal app saved.", true);
  };
  const test = card.querySelector("#pp-test");
  if (test) test.onclick = async () => {
    test.disabled = true;
    const t = await api("/api/paypal/credentials/test", { method: "POST", body: {} });

    setMsg(msg, t.data.ok ? (t.data.mock ? "Connected (simulated on this preview)." : "Connected to PayPal.") : t.data.error || "Could not connect.", !!t.data.ok);

    loadPayPalCard(gen, "PayPal disconnected.", true);
  };
}

const LOGO_MAX_EDGE = 384;
const LOGO_MAX_DATA_URL = 180000;
function imageFromUrl(url) {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error("decode"));
    img.src = url;
  });
}
async function loadImageFromFile(file) {
  if (typeof createImageBitmap === "function") {
    try { return await createImageBitmap(file); } catch { /* fall through */ }
  }
  let url = "";
  try {
    url = URL.createObjectURL(file);
    const img = await imageFromUrl(url);
    URL.revokeObjectURL(url);
    return img;
  } catch {
    if (url) try { URL.revokeObjectURL(url); } catch { /* ignore */ }
  }
  const dataUrl = await new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(String(reader.result || ""));
    reader.onerror = () => reject(new Error("decode"));
    reader.readAsDataURL(file);
  });
  return imageFromUrl(dataUrl);
}
async function compressLogoFile(file) {
  if (file.size > 15 * 1024 * 1024) throw new Error("too-large");
  const img = await loadImageFromFile(file);
  const canvas = document.createElement("canvas");
  const ctx = canvas.getContext("2d");
  if (!ctx) throw new Error("canvas");
  let edge = LOGO_MAX_EDGE;
  let dataUrl = "";
  for (let i = 0; i < 6; i++) {
    const w = img.naturalWidth || img.width;
    const h = img.naturalHeight || img.height;
    const scale = Math.min(1, edge / Math.max(w, h, 1));
    canvas.width = Math.max(1, Math.round(w * scale));
    canvas.height = Math.max(1, Math.round(h * scale));
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
    dataUrl = canvas.toDataURL("image/png");
    if (dataUrl.length <= LOGO_MAX_DATA_URL) {
      if (img && typeof img.close === "function") try { img.close(); } catch { /* ignore */ }
      return dataUrl;
    }
    edge = Math.floor(edge * 0.75);
  }
  if (img && typeof img.close === "function") try { img.close(); } catch { /* ignore */ }
  throw new Error("too-large");
}
function contrastText(bg) {
  const n = parseInt(String(bg).replace("#", ""), 16);
  const ch = (c) => { const s = c / 255; return s <= 0.04045 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4); };
  const L = Number.isFinite(n) ? 0.2126 * ch((n >> 16) & 255) + 0.7152 * ch((n >> 8) & 255) + 0.0722 * ch(n & 255) : 0;
  return (1.05 / (L + 0.05)) >= ((L + 0.05) / 0.05) ? "#f5f5f5" : "#111111";
}

function brandLum(hex) {
  const m = /^#?([0-9a-f]{6})$/i.exec(String(hex || "").trim());
  if (!m) return null;
  const n = parseInt(m[1], 16);
  const ch = (c) => { const s = c / 255; return s <= 0.04045 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4); };
  return 0.2126 * ch((n >> 16) & 255) + 0.7152 * ch((n >> 8) & 255) + 0.0722 * ch(n & 255);
}
function brandRatio(a, b) {
  const x = brandLum(a); const y = brandLum(b);
  if (x === null || y === null) return 0;
  return (Math.max(x, y) + 0.05) / (Math.min(x, y) + 0.05);
}
// Instant client-side check on the four main colors (the server re-checks every derived color on save).
function brandQuickProblems(simple) {
  const out = [];
  const keys = ["primary", "accent", "background", "text"];
  for (const k of keys) if (brandLum(simple[k]) === null) out.push(`Enter ${k} as a 6-digit hex color, like #a8842f.`);
  if (out.length) return out;
  const dark = simple.base === "dark";
  const textOnButton = Math.max(...(dark ? [simple.background, "#111111", simple.text, "#ffffff"] : [simple.text, "#ffffff"]).map((c) => brandRatio(simple.primary, c)));
  const checks = [
    ["Body text on the page background", brandRatio(simple.text, simple.background), 4.5],
    ["Large headings", brandRatio(simple.accent, simple.background), 3],
    ["Button against the page", brandRatio(simple.primary, simple.background), 3],
    ["Button text", textOnButton, 4.5],
  ];
  for (const [label, r, need] of checks) if (r < need) out.push(`${label} is hard to read (${r.toFixed(2)}:1, needs ${need}:1).`);
  return out;
}

function blobToDataUrl(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(String(reader.result || ""));
    reader.onerror = () => reject(new Error("logo"));
    reader.readAsDataURL(file);
  });
}

function adminTrialText(t) {
  if (!t) return "";
  if (t.state === "active" || t.state === "converted") return t.auto && t.convertedAt ? "Paying customer · converted from the 15-ride trial" : "Paying customer · no trial limits";
  if (t.state === "ended") {
    const why = t.endedReason === "rides" ? t.ridesUsed + " rides completed" : t.endedReason === "days" ? "time limit passed" : t.endedReason === "canceled" ? "canceled by the owner" : "ended by you";
    return "Free trial ended (" + why + ") · " + t.ridesUsed + " of " + t.rideLimit + " rides used";
  }
  const kind = t.auto ? "15-ride trial, card on file" : "Free trial";
  const extra = t.state === "converting" ? " · converting to Solo" : "";
  return kind + " · " + t.ridesUsed + " of " + t.rideLimit + " rides used" + (t.timeLimited ? ", " + (t.daysLeft <= 1 ? "last day" : t.daysLeft + " days left") + " (ends " + t.endsOn + ")" : "") + extra;
}

// Visible admin messages: never fail silently (Nov 10, 2026).
function setMsg(p, text, ok) {
  if (!p) return;
  p.style.color = ok ? "var(--green)" : "";
  p.textContent = text;
  if (p.scrollIntoView) p.scrollIntoView({ block: "nearest", behavior: "smooth" });
}
function adminMsg(box, text, ok) {
  if (!box) return;
  let p = box.querySelector(":scope > .admin-msg");
  if (!p) { p = document.createElement("p"); p.className = "err admin-msg"; p.setAttribute("role", "alert"); box.appendChild(p); }
  setMsg(p, text, ok);
}
async function busy(button, work) {
  if (!button || button.dataset.busy) return;
  button.dataset.busy = "1";
  button.disabled = true;
  try { await work(); }
  finally { delete button.dataset.busy; button.disabled = false; }
}

      editForm.onsubmit = (submit) => {
        submit.preventDefault();
        return busy(editForm.querySelector("[type=submit]"), async () => { try {
          const saved = await api("/api/platform/companies/" + id, {
            method: "PATCH",
            body: { name: editForm.elements.name.value, plan: editForm.elements.plan.value },
          });
          if (!saved.data.ok) { adminMsg(once, saved.data.error || "Could not save.", false); return; }
          route();
        } catch (e) { adminMsg(once, SOMETHING_WRONG(e), false); } });
      };
      editForm.elements.name.focus({ preventScroll: true });
      once.scrollIntoView({ block: "nearest", behavior: "smooth" });
      return;
    }
    if (button.dataset.act === "reset") {
      const owner = card.dataset.owner || "";
      once.innerHTML = `<form class="reset">
        <label>User email<input name="email" type="email" value="${esc(owner)}" required></label>
        <button class="btn full" type="submit">Create reset link</button>
      </form>`;
      const resetForm = once.querySelector("form");
      resetForm.onsubmit = (submit) => {
        submit.preventDefault();
        return busy(resetForm.querySelector("[type=submit]"), async () => { try {
          const result = await api("/api/platform/resets", { method: "POST", body: { email: resetForm.elements.email.value } });
          if (!result.data.ok) { adminMsg(once, result.data.error || "Could not create the link.", false); return; }
          const link = location.origin + "/#/start/" + result.data.inviteToken;
          once.innerHTML = `<p class="note">Copy this reset link now. It is shown once.</p><label>Reset link<input readonly value="${esc(link)}"></label>`;
        } catch (e) { adminMsg(once, SOMETHING_WRONG(e), false); } });
      };
      if (!owner) adminMsg(once, "This company has no owner yet. Enter the user's email.", false);
      resetForm.elements.email.focus({ preventScroll: true });
      once.scrollIntoView({ block: "nearest", behavior: "smooth" });
      return;
    }
    const path = button.dataset.act === "suspend" ? "suspend" : "reactivate";
    const result = await api("/api/platform/companies/" + id + "/" + path, { method: "POST", body: {} });
    if (!result.data.ok) { once.innerHTML = ""; adminMsg(once, result.data.error || "Could not update the company.", false); return; }
    route();
  };
}

const RS_BLANK_VEHICLE = () => ({ name: "", seats: "", base: "", perMile: "", minimum: "", hourlyRate: "", minHours: "", blockMinutes: "", blockPrice: "" });
const RS_BLANK_FARE = (n) => ({ place: "", placeKeywords: "", area: "", areaKeywords: "", prices: Array.from({ length: n }, () => ({ oneWay: "", roundTrip: "" })) });

function rsFromServer(enabled, sheet) {
  const s = sheet || { vehicles: [{ ...RS_BLANK_VEHICLE(), name: "Sedan", seats: "3" }], flatFares: [], fees: [], gratuityPercent: 0, roundTripFactor: 2, roundTo: 1, maxMiles: 100 };
  const names = s.vehicles.map((v) => v.name);
  return {
    enabled: !!enabled,
    vehicles: s.vehicles.map((v) => ({ ...RS_BLANK_VEHICLE(), ...v })),
    flatFares: (s.flatFares || []).map((f) => ({ ...f, prices: names.map((name) => ({ oneWay: "", roundTrip: "", ...((f.prices || {})[name] || {}) })) })),
    fees: (s.fees || []).map((f) => ({ label: f.label, kind: f.kind, appliesTo: f.appliesTo || "all", amount: f.amount })),
    gratuityPercent: String(s.gratuityPercent ?? 0),
    roundTripFactor: String(s.roundTripFactor ?? 2),
    roundTo: String(s.roundTo ?? 1),
    maxMiles: String(s.maxMiles ?? 100),
  };
}

function rsToServer(rs) {

    },
  };
}

function rsCollect(rs) {
  document.querySelectorAll("#rates [data-path]").forEach((el) => {
    const parts = el.dataset.path.split(".");
    let target = rs;
    for (let i = 0; i < parts.length - 1; i++) {
      const key = /^\d+$/.test(parts[i]) ? Number(parts[i]) : parts[i];
      if (target[key] == null) target[key] = /^\d+$/.test(parts[i + 1]) ? [] : {};
      target = target[key];
    }
    const last = parts[parts.length - 1];
    target[last] = el.type === "checkbox" ? el.checked : el.value;
  });
  return rs;
}

// Open a window in the click (sync) so iOS Safari / Android Chrome PWAs don't block Maps
// after the step save. If the pre-opened tab is missing, fall back to this window.
function openMapsWhenStepSaved(step) {
  if (!shouldAutoOpenMaps(step, false)) return { go() {}, cancel() {} };
  let win = null;
  try { win = window.open("about:blank", "_blank"); } catch (e) { win = null; }
  return {
    go(url) {
      if (!url) { this.cancel(); return; }
      if (win && !win.closed) {
        try { win.location = url; return; } catch (e) {}
      }
      location.href = url;
    },
    cancel() {
      if (win && !win.closed) { try { win.close(); } catch (e) {} }
    }
  };
}

function wireStopEditor(root) {
  if (!root) return;
  root.addEventListener("click", (event) => {
    const remove = event.target.closest("[data-remove-stop]");
    if (remove) { remove.closest(".stop-edit-row").remove(); syncStopEditor(root); return; }
    if (event.target.closest("[data-add-stop]")) {
      const rows = root.querySelector(".stop-edit-rows");
      const count = rows.querySelectorAll(".stop-edit-row").length;
      if (count >= MAX_STOPS) return;
      rows.insertAdjacentHTML("beforeend", stopEditRowHtml("", count));
      syncStopEditor(root);
      const inputs = rows.querySelectorAll("input");
      inputs[inputs.length - 1].focus();
    }
  });
  syncStopEditor(root);
}

function fareSuggestion(ride) {

const FARE_RE = /^\$?\d{1,3}(,?\d{3})*(\.\d{1,2})?$|^\$?\d+(\.\d{1,2})?$/;

async function renderRateSheet(gen) {
  setTabs(true);
  shell(`<p class="muted"><a href="#/settings">More</a></p><h1>Rates &amp; instant quotes</h1><div id="box" class="muted">Loading\u2026</div><div id="airbox"></div>`);
  const result = await api("/api/rate-sheet");
  if (gen !== renderGen) return;
  const box = document.querySelector("#box");
  if (!result.data.ok) { box.textContent = result.data.error || "Could not load your rates."; return; }
  const distanceOn = !!result.data.distanceLookup;
  let rs = rsFromServer(result.data.enabled, result.data.rateSheet);
  const draw = () => {
    box.classList.remove("muted");
    box.innerHTML = rsEditorHtml(rs, distanceOn);
    const on = (sel, fn) => { const el = document.querySelector(sel); if (el) el.onclick = () => { rsCollect(rs); fn(); draw(); }; };
    on("#add-vehicle", () => { rs.vehicles.push(RS_BLANK_VEHICLE()); rs.flatFares.forEach((f) => f.prices.push({ oneWay: "", roundTrip: "" })); });
    on("#add-fare", () => rs.flatFares.push(RS_BLANK_FARE(rs.vehicles.length)));
    on("#add-airport", () => rs.flatFares.push({ ...RS_BLANK_FARE(rs.vehicles.length), place: "ATL Airport", placeKeywords: "atl, hartsfield, atlanta airport, atl airport, hartsfield-jackson" }));
    on("#add-fee", () => rs.fees.push({ label: "", kind: "flat", appliesTo: "all", amount: "" }));
    document.querySelectorAll("[data-remove-vehicle]").forEach((b) => b.onclick = () => { rsCollect(rs); const i = Number(b.dataset.removeVehicle); rs.vehicles.splice(i, 1); rs.flatFares.forEach((f) => f.prices.splice(i, 1)); draw(); });
    document.querySelectorAll("[data-remove-fare]").forEach((b) => b.onclick = () => { rsCollect(rs); rs.flatFares.splice(Number(b.dataset.removeFare), 1); draw(); });
    document.querySelectorAll("[data-remove-fee]").forEach((b) => b.onclick = () => { rsCollect(rs); rs.fees.splice(Number(b.dataset.removeFee), 1); draw(); });
    document.querySelectorAll('#rates select[data-path$=".blockMinutes"]').forEach((sel) => sel.onchange = () => { rsCollect(rs); draw(); });
    const form = document.querySelector("#rates");
    form.onsubmit = async (event) => {
      event.preventDefault();
      rsCollect(rs);
      const button = form.querySelector("button[type=submit]");
      button.disabled = true;
      try {
        const saved = await api("/api/rate-sheet", { method: "PUT", body: rsToServer(rs) });
        if (gen !== renderGen) return;
        if (!saved.data.ok) { formError(saved.data.error || "Could not save your rates."); return; }
        rs = rsFromServer(saved.data.enabled, saved.data.rateSheet);
        draw();
        formError(rs.enabled ? "Saved. Clients now see estimates on your booking page." : "Saved. Estimates stay hidden until you turn them on.");
        document.querySelector("#err").style.color = "var(--green)";
      } finally {
        button.disabled = false;
      }
    };
    document.querySelector("#tryq").onsubmit = async (event) => {
      event.preventDefault();
      const out = document.querySelector("#tq-out");
      out.textContent = "Calculating\u2026";
      const next = await api("/api/rate-sheet/preview", { method: "POST", body: {
        tripType: document.querySelector("#tq-type").value,
        vehicle: document.querySelector("#tq-vehicle").value,
        pickup: document.querySelector("#tq-pickup").value,
        dropoff: document.querySelector("#tq-dropoff").value,
        hours: Number(document.querySelector("#tq-hours").value) || null,
      } });
      if (!next.data.ok) { out.textContent = next.data.error || "Could not get an estimate."; return; }
      const q = next.data.quote;
      if (!q) { out.textContent = "Choose a vehicle."; return; }
      if (!q.available) { out.textContent = "Price on request: " + q.message; return; }
      out.innerHTML = quoteBoxHtml({ total: (q.estimate ? "About " : "") + q.total, label: q.label + (q.miles ? ", " + q.miles + " mi" : ""), lines: q.lines, notes: q.notes });
    };
  };
  draw();
  renderAirportSettings(gen);
}

// Airport pickups on the booking page, passenger limits, and flight alerts (owner only).
// `notice` is shown after a save re-renders the card, so a tap on Save always gives visible feedback.
/* air-save-logic-start */
const AIR_SAVED = { preset: "Saved. ATL terminals are on your booking page.", off: "Saved. Airport options are off.", save: "Saved airport settings." };
function airSaveBody(kind, s, sedanValue, earlyValue) {
  if (kind === "preset") return { body: { preset: ["ATL"] } };
  if (kind === "off") return { body: { airports: { airports: [] } } };
  const sedan = Number(String(sedanValue == null ? "" : sedanValue).trim());
  if (!Number.isInteger(sedan) || sedan < 1 || sedan > 56) return { error: "Most passengers in a sedan must be a whole number from 1 to 56." };
  const early = Number(earlyValue);
  return { body: {
    capacities: { ...((s && s.capacities) || {}), sedan },
    flightRules: { ...((s && s.flightRules) || {}), earlyMinutes: early },
  } };
}
/* air-save-logic-end */

// Regular client schedules. Rides are never created automatically: each upcoming date waits here

const WEEKDAYS = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
function ordinalDay(n) {
  const v = n % 100;
  if (v >= 11 && v <= 13) return n + "th";
  return n + ({ 1: "st", 2: "nd", 3: "rd" }[n % 10] || "th");
}
function isOwnerView() {
  return !!(state.me && state.me.role === "owner" && !state.me.platformAdmin);
}
function paintScheduleBadge(count) {
  const button = tabs.querySelector('[data-go="#/rides"]');
  if (!button) return;
  let badge = button.querySelector(".tab-badge");
  const n = Number(count) || 0;
  if (!n) { if (badge) badge.remove(); button.removeAttribute("aria-label"); return; }
  if (!badge) { badge = document.createElement("span"); badge.className = "tab-badge"; button.appendChild(badge); }
  badge.textContent = n > 99 ? "99+" : String(n);
  button.setAttribute("aria-label", `Rides, ${n} regular trip${n === 1 ? "" : "s"} to confirm`);
}

// One click handler for any list of occurrence cards. `after` re-renders.
function wireOccurrenceList(root, gen, after) {
  root.addEventListener("click", async (event) => {
    const card = event.target.closest("[data-occ]");
    if (!card) return;
    const [sid, date] = card.dataset.occ.split("|");
    if (event.target.closest("[data-occ-edit]")) { location.hash = `#/regular/${sid}/${date}`; return; }
    const confirmBtn = event.target.closest("[data-occ-confirm]");
    const skipBtn = event.target.closest("[data-occ-skip]");
    if (!confirmBtn && !skipBtn) return;
    if (skipBtn && !window.confirm("Skip this date? No ride is created for it.")) return;
    const button = confirmBtn || skipBtn;
    button.disabled = true;
    const result = await api(`/api/schedules/${sid}/occurrences/${date}/${confirmBtn ? "confirm" : "skip"}`, { method: "POST", body: {} });
    if (gen !== renderGen) return;
    button.disabled = false;
    if (!result.data.ok) { formError(result.data.error || "Could not update that date."); return; }
    after();
  });
}

  const note = root.querySelector("#s-monthnote");

}

    api("/api/clients/" + clientId),

  ]);

// ---- Driver ride alerts (draft Nov 19, 2026) ----
// Locked phone / app closed: Web Push shows a system notification with the phone's default sound
// (web push has no custom sound on iPhone or Android). App open: the in-app chime + "You've got a ride"
// voice, a vibration, and a banner. iPhone needs Nudge added to the Home Screen (iOS 16.4+) for push.
const rideAlerts = { config: null, audio: null, unlocked: false, seen: null, pollTimer: 0, alerted: new Set(), synced: false };
const ALERT_VIBRATE = [300, 120, 300, 120, 600];

function isDriverView() { return !!(state.me && state.me.role === "driver"); }
function alertIsIos() {
  return /iPhone|iPad|iPod/.test(navigator.userAgent) || (navigator.platform === "MacIntel" && navigator.maxTouchPoints > 1);
}
function alertIsStandalone() {
  return (window.matchMedia && window.matchMedia("(display-mode: standalone)").matches) || navigator.standalone === true;
}
function alertPushSupported() {
  return "serviceWorker" in navigator && "PushManager" in window && "Notification" in window && window.isSecureContext !== false;
}
function alertPlatform() {
  if (alertIsIos()) return "ios";
  if (/Android/.test(navigator.userAgent)) return "android";
  return "desktop";
}
function alertKeyBytes(b64) {
  const pad = "=".repeat((4 - (b64.length % 4)) % 4);
  const raw = atob((b64 + pad).replace(/-/g, "+").replace(/_/g, "/"));
  return Uint8Array.from(raw, (c) => c.charCodeAt(0));
}

function alertAudio() {
  if (rideAlerts.audio) return rideAlerts.audio;
  const audio = new Audio();
  audio.preload = "auto";
  audio.src = audio.canPlayType("audio/mpeg") ? "/sounds/ride-alert.mp3" : "/sounds/ride-alert.ogg";
  rideAlerts.audio = audio;
  return audio;
}
// Browsers only allow sound after a tap. The first tap anywhere unlocks it silently.
function unlockAlertSound() {
  if (rideAlerts.unlocked || !isDriverView()) return;
  const audio = alertAudio();
  audio.muted = true;
  const done = () => { audio.pause(); audio.currentTime = 0; audio.muted = false; };
  const played = audio.play();
  if (played && played.then) played.then(() => { rideAlerts.unlocked = true; done(); }).catch(() => { audio.muted = false; });
  else { rideAlerts.unlocked = true; done(); }
}
document.addEventListener("pointerdown", unlockAlertSound, { capture: true });

function playRideAlert() {
  const audio = alertAudio();
  try { audio.currentTime = 0; } catch {}
  audio.muted = false;
  const played = audio.play();
  if (played && played.catch) played.catch(() => {});
  if (navigator.vibrate) { try { navigator.vibrate(ALERT_VIBRATE); } catch {} }
}

function showRideAlertBanner(alert) {
  let banner = document.querySelector("#ride-alert-banner");
  if (!banner) {
    banner = document.createElement("div");
    banner.id = "ride-alert-banner";
    banner.className = "ride-alert-banner";
    banner.setAttribute("role", "alert");
    document.body.appendChild(banner);
  }
  const open = alert.url || (alert.rideId ? "/#/rides/" + alert.rideId : "/#/rides");
  banner.dataset.kind = alert.kind || "assigned";
  banner.innerHTML = `<div class="rab-text"><strong>${esc(alert.title || "New ride assigned")}</strong><span>${esc(alert.body || "")}</span></div>
    <div class="rab-actions"><button class="btn" type="button" id="rab-open">Open</button><button class="btn ghost" type="button" id="rab-close" aria-label="Dismiss">✕</button></div>`;
  banner.querySelector("#rab-open").onclick = () => { banner.remove(); location.hash = open.replace(/^\/?/, "").replace(/^#?/, "#").replace(/^##/, "#"); };
  banner.querySelector("#rab-close").onclick = () => banner.remove();
}

function onRideAlert(alert) {
  if (!isDriverView() || !alert) return;
  if (alert.rideId && alert.kind === "assigned") {
    const key = alert.kind + ":" + alert.rideId;
    if (rideAlerts.alerted.has(key)) return;
    rideAlerts.alerted.add(key);
    if (rideAlerts.seen) rideAlerts.seen.add(alert.rideId);
  }
  playRideAlert();
  showRideAlertBanner(alert);
  if (routeName().name === "rides" && !routeName().id) route();
}

if ("serviceWorker" in navigator) {
  navigator.serviceWorker.addEventListener("message", (event) => {
    const data = event.data || {};
    if (data.type === "nudge-ride-alert") onRideAlert(data.alert);
    if (data.type === "nudge-open" && typeof data.url === "string") {
      const hash = new URL(data.url, location.origin).hash;
      if (hash) location.hash = hash;
    }
  });
}

// Backup for phones without push (or iPhone in Safari): while the app is open, check every 30 s.
function alertDay(iso) {
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso || "");
  if (!m) return iso || "";
  return new Date(Date.UTC(+m[1], +m[2] - 1, +m[3])).toLocaleDateString("en-US", { timeZone: "UTC", weekday: "short", month: "short", day: "numeric" }).replace(",", "");
}
function rideAlertKey(ride) { return ride.status === "cancelled" || ride.status === "completed" ? "" : ride.id; }
async function pollDriverRides() {
  if (!isDriverView() || document.visibilityState !== "visible") return;
  const result = await api("/api/rides");
  if (!result.data.ok) return;
  const live = (result.data.rides || []).map((ride) => ({ ride, key: rideAlertKey(ride) })).filter((x) => x.key);
  if (!rideAlerts.seen) { rideAlerts.seen = new Set(live.map((x) => x.key)); return; }
  const fresh = live.filter((x) => !rideAlerts.seen.has(x.key));
  for (const x of live) rideAlerts.seen.add(x.key);
  if (!fresh.length) return;
  const ride = fresh[0].ride;
  onRideAlert({ kind: "assigned", rideId: ride.id, title: fresh.length > 1 ? `${fresh.length} new rides assigned` : "New ride assigned", body: `Pickup ${alertDay(ride.date)}, ${time12(ride.time)}`, url: "/#/rides/" + ride.id });
}
function startDriverPolling() {
  if (rideAlerts.pollTimer || !isDriverView()) return;
  pollDriverRides();
  rideAlerts.pollTimer = setInterval(pollDriverRides, 30000);
}
function stopDriverPolling() {
  if (rideAlerts.pollTimer) clearInterval(rideAlerts.pollTimer);
  rideAlerts.pollTimer = 0;
  rideAlerts.seen = null;
}
document.addEventListener("visibilitychange", () => { if (document.visibilityState === "visible") pollDriverRides(); });

async function rideAlertConfig() {
  if (rideAlerts.config) return rideAlerts.config;
  const result = await api("/api/push/config");
  rideAlerts.config = result.data.ok ? result.data : { enabled: false };
  return rideAlerts.config;
}
async function currentPushSubscription() {
  if (!alertPushSupported()) return null;
  try {
    const reg = await navigator.serviceWorker.ready;
    return await reg.pushManager.getSubscription();
  } catch { return null; }
}
async function saveSubscription(sub) {
  const body = { ...sub.toJSON(), platform: alertPlatform() };
  return api("/api/push/subscribe", { method: "POST", body });
}
// Re-register once per session so a rotated subscription keeps working.
async function syncRideAlerts() {
  if (rideAlerts.synced || !isDriverView() || !alertPushSupported() || Notification.permission !== "granted") return;
  rideAlerts.synced = true;
  const config = await rideAlertConfig();
  if (!config.enabled) return;
  const sub = await currentPushSubscription();
  if (sub) await saveSubscription(sub);
}
async function turnOnRideAlerts() {
  unlockAlertSound();
  const config = await rideAlertConfig();
  if (!config.enabled) return "Ride alerts are not set up yet.";
  const permission = await Notification.requestPermission();
  if (permission !== "granted") return permission === "denied" ? "" : "Tap Allow when your phone asks, so ride alerts can show.";
  let sub;
  try {
    const reg = await navigator.serviceWorker.ready;
    sub = await reg.pushManager.getSubscription();
    if (sub && sub.options && sub.options.applicationServerKey) {
      const current = new Uint8Array(sub.options.applicationServerKey);
      const want = alertKeyBytes(config.publicKey);
      if (current.length !== want.length || current.some((b, i) => b !== want[i])) { await sub.unsubscribe(); sub = null; }
    }
    if (!sub) sub = await reg.pushManager.subscribe({ userVisibleOnly: true, applicationServerKey: alertKeyBytes(config.publicKey) });
  } catch {
    return "This phone could not turn on alerts. Check that notifications are allowed for Nudge and try again.";
  }
  const saved = await saveSubscription(sub);
  if (!saved.data.ok) return saved.data.error || "Could not save ride alerts.";
  rideAlerts.synced = true;
  return "";
}
async function turnOffRideAlerts() {
  const sub = await currentPushSubscription();
  if (!sub) return;
  await api("/api/push/unsubscribe", { method: "POST", body: { endpoint: sub.endpoint } });
  try { await sub.unsubscribe(); } catch {}
}

async function route() {
  locWatch.on = false;
  locWatch.rideId = "";
  locStopWatch();
  const gen = ++renderGen;
  const { name, id } = routeName();
  setTag();
  paintGear();
  if (name === "where" && id) return renderWhere(id, gen);
  if (state.me && state.me.platformAdmin && name !== "admin" && name !== "login" && name !== "start") {
    location.hash = "#/admin";
    return;
  }
  if (name === "admin") {
    if (!state.me || !state.me.platformAdmin) {
      location.hash = state.me ? "#/rides" : "#/login";
      return;
    }
    const adminParts = routeName().parts;
    if (id === "brand" && /^[0-9a-f-]{36}$/.test(adminParts[2] || "")) return renderBrandSettings(gen, adminParts[2]);
    return renderAdmin(gen);
  }
  if (billingLocked() && name !== "login" && name !== "start" && name !== "welcome") return renderLocked();
  if (state.me && state.me.role === "driver" && name !== "rides" && name !== "login" && name !== "start") {
    location.hash = "#/rides";
    return;
  }
  if (!state.me && name !== "login" && name !== "start") return renderLogin();
  if (name === "login") return renderLogin();
  if (name === "start") return renderStart();
  if (name === "welcome") return renderWelcome(gen);
  const parts = routeName().parts;
  if (isOwnerView() && name !== "rides") refreshScheduleBadge();
  if (name === "clients" && id && id !== "new" && parts[2] === "schedule" && parts[3]) return renderScheduleForm(id, parts[3] === "new" ? "" : parts[3], gen);
  if (name === "regular" && id && /^\d{4}-\d{2}-\d{2}$/.test(parts[2] || "")) return renderOccurrenceEdit(id, parts[2], gen);
  if (name === "clients" && id === "new") return renderClientForm("", gen);
  if (name === "clients" && id) return renderClientForm(id, gen);
  if (name === "clients") return renderClients(gen);
  if (name === "new") return renderNewRide(gen);
  if (name === "rides" && id) return renderRide(id, gen);
  if (name === "invoices" && id) return renderInvoice(id, gen);
  if (name === "settings" && id === "rates") return renderRateSheet(gen);
  if (name === "settings" && id === "brand") return renderBrandSettings(gen);
  if (name === "profile" && id === "cancel") return renderCancelTrial(gen);
  if (name === "profile") return renderProfile(gen);
  if (name === "settings") return renderSettings(gen);
  return renderRides(gen);
}
