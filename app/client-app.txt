const esc = (v) => String(v == null ? "" : v).replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));

function longIssued(iso) {
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso || "");
  if (!match) return iso || "";
  const dt = new Date(Date.UTC(+match[1], +match[2] - 1, +match[3]));
  const dow = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"][dt.getUTCDay()];
  const mon = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"][+match[2] - 1];
  return `${dow}, ${mon} ${+match[3]}, ${match[1]}`;
}
function time12(value) {
  const match = /^(\d{2}):(\d{2})$/.exec(value || "");
  if (!match) return value || "";
  let hour = +match[1];
  const suffix = hour >= 12 ? "PM" : "AM";
  hour = hour % 12 || 12;
  return `${hour}:${match[2]} ${suffix}`;
}

const locWatch = { id: null, timer: null, on: false, rideId: "" };

function locStopWatch() {
  if (locWatch.id != null && navigator.geolocation) navigator.geolocation.clearWatch(locWatch.id);
  locWatch.id = null;
  if (locWatch.timer) clearInterval(locWatch.timer);
  locWatch.timer = null;
}

function locSendFix(pos) {
  if (!locWatch.rideId) return Promise.resolve();
  return api("/api/location", {
    method: "POST",
    body: {
      rideId: locWatch.rideId,
      lat: pos.coords.latitude,
      lng: pos.coords.longitude,
      acc: typeof pos.coords.accuracy === "number" ? pos.coords.accuracy : null,
      ts: pos.timestamp || Date.now(),
    },
  });
}

function locStartWatch() {
  locStopWatch();
  if (!navigator.geolocation || window.isSecureContext === false) return;
  const opt = { enableHighAccuracy: false, maximumAge: 30000, timeout: 30000 };
  locWatch.id = navigator.geolocation.watchPosition((pos) => { locSendFix(pos); }, () => {}, opt);
  locWatch.timer = setInterval(() => {
    if (!locWatch.on || document.visibilityState !== "visible" || !navigator.geolocation) return;
    navigator.geolocation.getCurrentPosition((pos) => { locSendFix(pos); }, () => {}, opt);
  }, 45000);
}

document.addEventListener("visibilitychange", () => {
  if (!locWatch.on) return;
  if (document.visibilityState !== "visible") locStopWatch();
  else locStartWatch();
});

function whereToken() {
  const match = /^#\/where\/([A-Za-z0-9_-]{20,})$/.exec(location.hash || "");
  return match ? match[1] : "";
}

function routeName() {
  const hash = location.hash || "#/rides";
  const parts = hash.replace(/^#/, "").split("/").filter(Boolean);
  return { parts, name: parts[0] || "rides", id: parts[1] || "" };
}

async function api(path, options = {}) {
  const headers = { "X-Nudge-Client": "1" };
  if (options.body) headers["Content-Type"] = "application/json";
  if (options.auth !== false && state.me && options.method && options.method !== "GET") headers["X-CSRF-Token"] = state.me.csrf;
  let response;
  try {
    response = await fetch(path, {
      method: options.method || "GET",
      headers,
      body: options.body ? JSON.stringify(options.body) : undefined,
      credentials: "same-origin",
    });
  } catch {
    return { status: 0, data: { ok: false, error: "No connection. Check your signal and try again." } };
  }
  const data = await response.json().catch(() => ({ ok: false, error: "Unexpected response (" + response.status + ")." }));
  if (!response.ok && data && typeof data === "object" && data.ok !== false) { data.ok = false; data.error = data.error || "Request failed (" + response.status + ")."; }
  if (response.status === 403 && data && data.billingLocked && state.me && !state.me.platformAdmin) {
    state.me.access = { ...(state.me.access || {}), state: "locked", locked: true, banner: null, lockedView: data.lockedView || null };
    setTimeout(() => renderLocked(), 0);
  }
  if (response.status === 401 && path !== "/api/login" && path !== "/api/invites/accept" && path !== "/api/driver-login") {
    const hadSession = !!state.me;
    state.me = null;
    const screen = (location.hash || "").replace(/^#\//, "").split("/")[0];
    if (hadSession || (screen && screen !== "login" && screen !== "start")) location.hash = "#/login";
  }
  return { status: response.status, data };
}

function formError(message) {
  let el = document.querySelector("#err");
  if (!el) {
    el = document.createElement("p");
    el.className = "err";
    el.id = "err";
    const form = document.querySelector("#f") || document.querySelector("#view");
    const button = form && form.querySelector("button[type=submit]");
    if (button) button.insertAdjacentElement("beforebegin", el);
    else if (form) form.appendChild(el);
    else return;
  }
  el.style.color = "";
  el.textContent = message || "";
}

const INVITE_TOKEN_KEY = "nudge_invite_token";

function rememberInviteToken(token) {
  state.inviteToken = token || "";
  try {
    if (token) sessionStorage.setItem(INVITE_TOKEN_KEY, token);
    else sessionStorage.removeItem(INVITE_TOKEN_KEY);
  } catch {}
}

function loadInviteToken() {
  if (state.inviteToken) return state.inviteToken;
  try { state.inviteToken = sessionStorage.getItem(INVITE_TOKEN_KEY) || ""; }
  catch { state.inviteToken = state.inviteToken || ""; }
  return state.inviteToken;
}

function clearInviteToken() {
  state.inviteToken = "";
  try { sessionStorage.removeItem(INVITE_TOKEN_KEY); } catch {}
}

async function boot() {
  const start = /^#\/start\/([A-Za-z0-9_-]+)$/.exec(location.hash);
  if (start) {
    rememberInviteToken(start[1]);
    history.replaceState(null, "", location.pathname + location.search + "#/start");
  } else if (location.hash === "#/start") {
    loadInviteToken();
  }
  if ("serviceWorker" in navigator) navigator.serviceWorker.register("/sw.js").catch(() => {});
  const me = await api("/api/me");
  if (me.data.ok) {
    state.me = me.data;
    setTag();
    if (location.hash === "" || location.hash === "#/login" || location.hash === "#/start") {
      location.hash = me.data.platformAdmin ? "#/admin" : "#/rides";
    }
    await route();
    return;
  }
  if (whereToken()) return route();
  if (location.hash === "#/start") return renderStart();
  location.hash = "#/login";
  renderLogin();
}

function startAcceptError(data) {
  const error = data && data.error || "";
  if (/invalid or already used/i.test(error)) {
    return "This setup link has expired or was already used. Ask your Nudge Dispatch contact for a new link.";
  }
  return error || "Could not save the password.";
}

function updateStartHint() {
  const password = document.querySelector("#password");
  const again = document.querySelector("#again");
  const hint = document.querySelector("#pw-hint");
  if (!password || !again || !hint) return;
  const a = password.value;
  const b = again.value;
  hint.style.color = "";
  hint.textContent = "";
  if (!a && !b) return;
  if (a.length < 12) {
    hint.style.color = "var(--red)";
    hint.textContent = "Use at least 12 characters.";
    return;
  }
  if (a.length > 200) {
    hint.style.color = "var(--red)";
    hint.textContent = "Use at most 200 characters.";
    return;
  }
  if (b && a !== b) {
    hint.style.color = "var(--red)";
    hint.textContent = "Those passwords do not match.";
    return;
  }
  if (b && a === b) {
    hint.style.color = "var(--green)";
    hint.textContent = "Passwords match.";
  }
}

  loadInviteToken();

      <p class="muted" id="pw-hint"></p>
      <p class="err" id="err"></p>
      <button class="btn full" type="submit">Save password and open my rides</button>
    </form>
  </section>`);
  const form = document.querySelector("#f");
  const passwordInput = document.querySelector("#password");
  const againInput = document.querySelector("#again");
  const button = form.querySelector("button[type=submit]");
  passwordInput.addEventListener("input", updateStartHint);
  againInput.addEventListener("input", updateStartHint);
  if (!state.inviteToken) formError("Open the one-time setup link you were given.");
  form.onsubmit = async (event) => {
    event.preventDefault();
    const password = passwordInput.value;
    const again = againInput.value;
    updateStartHint();
    if (!password) return formError("Enter a password.");
    if (password.length < 12 || password.length > 200) return formError("Use a password between 12 and 200 characters.");
    if (password !== again) return formError("Those passwords do not match.");
    if (!state.inviteToken) return formError("Open the one-time setup link you were given.");
    button.disabled = true;
    try {
      const acceptBody = { token: state.inviteToken, password };
      const result = await api("/api/invites/accept", { method: "POST", auth: false, body: acceptBody });
      if (!result.data.ok) return formError(startAcceptError(result.data));
      clearInviteToken();
      state.me = result.data;
      location.hash = "#/rides";
    } catch {
      formError("Could not save the password. Check your connection and try again.");
    } finally {
      button.disabled = false;
    }
  };
}

async function loadTrial() {
  state.trial = null;
  state.trialBanner = null;
  if (!state.me || state.me.role !== "owner" || state.me.platformAdmin) return null;
  const result = await api("/api/trial");

}
function trialClosed() {
  return !!(state.trial && state.trial.open === false);
}

      <h2>${esc(b.title)}</h2>
      <p class="trial-line">${esc(b.line)}</p>

      <p class="sub"><a href="#/settings">Export your rides and clients</a> any time from More.</p>

function rideCardHtml(ride, pending) {
  const when = `${esc(ride.date)} · ${esc(time12(ride.time))}`;
  const fare = ride.status === "requested" || !ride.fare || ride.fare === "$0.00" ? "" : " · " + esc(ride.fare);
  const phone = ride.clientPhone ? `<div class="sub">${esc(ride.clientPhone)}</div>` : "";
  const notes = pending && ride.notes ? `<div class="sub">${esc(ride.notes)}</div>` : "";
  const trip = (tripLine(ride) ? `<div class="sub">${esc(tripLine(ride))}</div>` : "") + airportHtml(ride, "div");
  const owner = !(state.me && state.me.role === "driver");
  const quote = pending && owner ? (ride.quote ? quoteBoxHtml(ride.quote) : `<div class="sub">No estimate was shown to the client.</div>`) : "";
  const actions = pending && owner ? `${trialClosed() ? trialPausedHtml() : approveBlockHtml(ride, `data-fare="${esc(ride.id)}"`, `data-approve="${esc(ride.id)}"`)}<div class="pend-actions"><button class="btn ghost" type="button" data-decline="${esc(ride.id)}">Decline</button></div>` : "";
  return `<div class="list-btn${pending ? " pending-request card" : ""}" data-id="${esc(ride.id)}">
      <div class="row"><span class="title">${esc(ride.clientName)}</span><span class="badge ${esc(ride.status)}">${esc(rideBadgeText(ride))}</span></div>
      <div class="sub when">${when}${fare}</div>
      <div class="sub">${esc(ride.pickup)} → ${esc(ride.dropoff)}</div>
      ${ride.driverName ? `<div class="sub">Driver: ${esc(ride.driverName)}</div>` : ""}
      ${trip}${phone}${notes}${quote}${actions}
    </div>`;
}

      if (!next.data.ok) { formError(next.data.error || "Could not update the request."); return; }

      return;
    }
    const button = event.target.closest("[data-id]");
    if (button) location.hash = "#/rides/" + button.dataset.id;
  };
}

  };

function download(name, contents, type) {
  const blob = new Blob([contents], { type: type + ";charset=utf-8" });
  const link = document.createElement("a");
  link.href = URL.createObjectURL(blob);
  link.download = name;
  document.body.appendChild(link);
  link.click();
  link.remove();
  setTimeout(() => URL.revokeObjectURL(link.href), 30000);
}

function paintDrivers(drivers, flash) {
  const list = document.querySelector("#driver-list");
  if (!list) return;
  if (!drivers) {
    list.innerHTML = `<p class="sub">Could not load drivers.</p>`;
    return;
  }
  const people = drivers.filter((member) => member.role === "driver");
  list.classList.toggle("muted", people.length === 0);
  list.innerHTML = people.length
    ? people.map((member) => driverCard(member, flash)).join("")
    : `<p class="sub">No drivers yet. Add one above.</p>`;
  for (const form of list.querySelectorAll(".edit-driver")) {
    form.onsubmit = async (event) => {
      event.preventDefault();
      const id = form.dataset.id;
      const err = form.querySelector(".err");
      const next = await api("/api/drivers/" + id, {
        method: "PATCH",
        body: {
          name: form.querySelector("[name=name]").value,
          vehicle: form.querySelector("[name=vehicle]").value,
          plate: form.querySelector("[name=plate]").value,
        },
      });
      if (!next.data.ok) {
        if (err) { err.classList.remove("ok"); err.textContent = next.data.error || "Could not save the driver."; }
        return;
      }
      await refreshDrivers({ id, kind: "saved" });
    };
  }
  for (const form of list.querySelectorAll(".passcode-driver")) {
    form.onsubmit = async (event) => {
      event.preventDefault();
      const id = form.dataset.id;
      const err = form.querySelector(".err");
      const next = await api("/api/drivers/" + id + "/passcode", {
        method: "PUT",
        body: { passcode: form.querySelector("[name=passcode]").value },
      });
      if (!next.data.ok) {
        if (err) { err.classList.remove("ok"); err.textContent = next.data.error || "Could not save the passcode."; }
        return;
      }
      const typed = form.querySelector("[name=passcode]").value;
      await refreshDrivers({ id, kind: "passcode", message: "Passcode saved. Tell the driver in person. It is not texted." });
      const input = document.querySelector(`[data-passcode-for="${CSS.escape(id)}"]`);
      if (input) input.value = typed;
    };
    const clear = form.querySelector("[data-clear]");
    if (clear) clear.onclick = async () => {
      const id = form.dataset.id;
      const err = form.querySelector(".err");
      const next = await api("/api/drivers/" + id + "/passcode", { method: "DELETE" });
      if (!next.data.ok) {
        if (err) { err.classList.remove("ok"); err.textContent = next.data.error || "Could not remove the passcode."; }
        return;
      }
      await refreshDrivers({ id, kind: "passcode", message: "Passcode removed. This driver cannot sign in until you set a new one." });
    };
  }
  for (const row of list.querySelectorAll(".driver-row")) {
    const remove = row.querySelector("[data-remove]");
    const confirm = row.querySelector("[data-confirm]");
    const cancel = row.querySelector("[data-cancel-delete]");
    const yes = row.querySelector("[data-confirm-delete]");
    if (remove && confirm) remove.onclick = () => {
      remove.classList.add("hidden");
      confirm.classList.remove("hidden");
    };
    if (cancel && confirm && remove) cancel.onclick = () => {
      confirm.classList.add("hidden");
      remove.classList.remove("hidden");
    };
    if (yes) yes.onclick = async () => {
      const id = row.dataset.id;
      const err = confirm.querySelector(".err");
      yes.disabled = true;
      const next = await api("/api/drivers/" + id, { method: "DELETE" });
      if (!next.data.ok) {
        yes.disabled = false;
        if (err) err.textContent = next.data.error || "Could not remove the driver.";
        return;
      }
      const n = Number(next.data.unassignedRides) || 0;
      const note = document.querySelector("#driver-err");
      if (note) {
        note.classList.add("ok");
        note.textContent = n === 0
          ? "Driver removed. They cannot sign in. Completed rides still show their name."
          : "Driver removed. " + n + " upcoming ride" + (n === 1 ? " is" : "s are") + " unassigned.";
      }
      await refreshDrivers();
    };
  }
  if (flash && flash.focusId) {
    const input = list.querySelector(`[data-passcode-for="${CSS.escape(flash.focusId)}"]`);
    const row = input && input.closest(".driver-row");
    if (row) row.scrollIntoView({ block: "nearest" });
    if (input) input.focus();
  }
}

async function refreshDrivers(flash) {
  const listed = await api("/api/drivers");
  const list = document.querySelector("#driver-list");
  if (!list) return listed;
  if (!listed.data.ok) {
    list.innerHTML = `<p class="sub">${esc(listed.data.error || "Could not load drivers.")}</p>`;
    return listed;
  }
  paintDrivers(listed.data.drivers || [], flash);
  return listed;
}

function blobToDataUrl(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(String(reader.result || ""));
    reader.onerror = () => reject(new Error("logo"));
    reader.readAsDataURL(file);
  });
}

async function busy(button, work) {
  if (!button || button.dataset.busy) return;
  button.dataset.busy = "1";
  button.disabled = true;
  try { await work(); }
  finally { delete button.dataset.busy; button.disabled = false; }
}

const RS_BLANK_VEHICLE = () => ({ name: "", seats: "", base: "", perMile: "", minimum: "", hourlyRate: "", minHours: "", blockMinutes: "", blockPrice: "" });
const RS_BLANK_FARE = (n) => ({ place: "", placeKeywords: "", area: "", areaKeywords: "", prices: Array.from({ length: n }, () => ({ oneWay: "", roundTrip: "" })) });

function rsFromServer(enabled, sheet) {
  const s = sheet || { vehicles: [{ ...RS_BLANK_VEHICLE(), name: "Sedan", seats: "3" }], flatFares: [], fees: [], gratuityPercent: 0, roundTripFactor: 2, roundTo: 1, maxMiles: 100 };
  const names = s.vehicles.map((v) => v.name);
  return {
    enabled: !!enabled,
    vehicles: s.vehicles.map((v) => ({ ...RS_BLANK_VEHICLE(), ...v })),
    flatFares: (s.flatFares || []).map((f) => ({ ...f, prices: names.map((name) => ({ oneWay: "", roundTrip: "", ...((f.prices || {})[name] || {}) })) })),
    fees: (s.fees || []).map((f) => ({ label: f.label, kind: f.kind, appliesTo: f.appliesTo || "all", amount: f.amount })),
    gratuityPercent: String(s.gratuityPercent ?? 0),
    roundTripFactor: String(s.roundTripFactor ?? 2),
    roundTo: String(s.roundTo ?? 1),
    maxMiles: String(s.maxMiles ?? 100),
  };
}

function rsToServer(rs) {
  return {
    enabled: rs.enabled,
    rateSheet: {
      vehicles: rs.vehicles,
      flatFares: rs.flatFares.map((f) => ({
        place: f.place, placeKeywords: f.placeKeywords, area: f.area, areaKeywords: f.areaKeywords,
        prices: Object.fromEntries(rs.vehicles.map((v, i) => [v.name, f.prices[i] || { oneWay: "", roundTrip: "" }]).filter(([name]) => name)),
      })),
      fees: rs.fees,
      gratuityPercent: rs.gratuityPercent,
      roundTripFactor: rs.roundTripFactor,
      roundTo: Number(rs.roundTo),
      maxMiles: rs.maxMiles,
    },
  };
}

function rsCollect(rs) {
  document.querySelectorAll("#rates [data-path]").forEach((el) => {
    const parts = el.dataset.path.split(".");
    let target = rs;
    for (let i = 0; i < parts.length - 1; i++) {
      const key = /^\d+$/.test(parts[i]) ? Number(parts[i]) : parts[i];
      if (target[key] == null) target[key] = /^\d+$/.test(parts[i + 1]) ? [] : {};
      target = target[key];
    }
    const last = parts[parts.length - 1];
    target[last] = el.type === "checkbox" ? el.checked : el.value;
  });
  return rs;
}

// Keep in sync with app/progress-nav.js shouldAutoOpenMaps.
function shouldAutoOpenMaps(step, undo) {
  return undo !== true && (step === "En route" || step === "Picked up");
}

// Open a window in the click (sync) so iOS Safari / Android Chrome PWAs don't block Maps
// after the step save. If the pre-opened tab is missing, fall back to this window.
function openMapsWhenStepSaved(step) {
  if (!shouldAutoOpenMaps(step, false)) return { go() {}, cancel() {} };
  let win = null;
  try { win = window.open("about:blank", "_blank"); } catch (e) { win = null; }
  return {
    go(url) {
      if (!url) { this.cancel(); return; }
      if (win && !win.closed) {
        try { win.location = url; return; } catch (e) {}
      }
      location.href = url;
    },
    cancel() {
      if (win && !win.closed) { try { win.close(); } catch (e) {} }
    }
  };
}

    const block = input.closest(".approve-price");

    const sync = () => { button.disabled = !fareOk(input.value); };
    input.addEventListener("input", sync);
    sync();
    const use = block.nextElementSibling && block.nextElementSibling.querySelector(".use-est");
    if (use) use.addEventListener("click", (event) => { event.preventDefault(); event.stopPropagation(); input.value = use.dataset.use; sync(); input.focus(); });
  });
}

async function renderRateSheet(gen) {
  setTabs(true);
  shell(`<p class="muted"><a href="#/settings">More</a></p><h1>Rates &amp; instant quotes</h1><div id="box" class="muted">Loading\u2026</div><div id="airbox"></div>`);
  const result = await api("/api/rate-sheet");
  if (gen !== renderGen) return;
  const box = document.querySelector("#box");
  if (!result.data.ok) { box.textContent = result.data.error || "Could not load your rates."; return; }
  const distanceOn = !!result.data.distanceLookup;
  let rs = rsFromServer(result.data.enabled, result.data.rateSheet);
  const draw = () => {
    box.classList.remove("muted");
    box.innerHTML = rsEditorHtml(rs, distanceOn);
    const on = (sel, fn) => { const el = document.querySelector(sel); if (el) el.onclick = () => { rsCollect(rs); fn(); draw(); }; };
    on("#add-vehicle", () => { rs.vehicles.push(RS_BLANK_VEHICLE()); rs.flatFares.forEach((f) => f.prices.push({ oneWay: "", roundTrip: "" })); });
    on("#add-fare", () => rs.flatFares.push(RS_BLANK_FARE(rs.vehicles.length)));
    on("#add-airport", () => rs.flatFares.push({ ...RS_BLANK_FARE(rs.vehicles.length), place: "ATL Airport", placeKeywords: "atl, hartsfield, atlanta airport, atl airport, hartsfield-jackson" }));
    on("#add-fee", () => rs.fees.push({ label: "", kind: "flat", appliesTo: "all", amount: "" }));
    document.querySelectorAll("[data-remove-vehicle]").forEach((b) => b.onclick = () => { rsCollect(rs); const i = Number(b.dataset.removeVehicle); rs.vehicles.splice(i, 1); rs.flatFares.forEach((f) => f.prices.splice(i, 1)); draw(); });
    document.querySelectorAll("[data-remove-fare]").forEach((b) => b.onclick = () => { rsCollect(rs); rs.flatFares.splice(Number(b.dataset.removeFare), 1); draw(); });
    document.querySelectorAll("[data-remove-fee]").forEach((b) => b.onclick = () => { rsCollect(rs); rs.fees.splice(Number(b.dataset.removeFee), 1); draw(); });
    document.querySelectorAll('#rates select[data-path$=".blockMinutes"]').forEach((sel) => sel.onchange = () => { rsCollect(rs); draw(); });
    const form = document.querySelector("#rates");
    form.onsubmit = async (event) => {
      event.preventDefault();
      rsCollect(rs);
      const button = form.querySelector("button[type=submit]");
      button.disabled = true;
      try {
        const saved = await api("/api/rate-sheet", { method: "PUT", body: rsToServer(rs) });
        if (gen !== renderGen) return;
        if (!saved.data.ok) { formError(saved.data.error || "Could not save your rates."); return; }
        rs = rsFromServer(saved.data.enabled, saved.data.rateSheet);
        draw();
        formError(rs.enabled ? "Saved. Clients now see estimates on your booking page." : "Saved. Estimates stay hidden until you turn them on.");
        document.querySelector("#err").style.color = "var(--green)";
      } finally {
        button.disabled = false;
      }
    };
    document.querySelector("#tryq").onsubmit = async (event) => {
      event.preventDefault();
      const out = document.querySelector("#tq-out");
      out.textContent = "Calculating\u2026";
      const next = await api("/api/rate-sheet/preview", { method: "POST", body: {
        tripType: document.querySelector("#tq-type").value,
        vehicle: document.querySelector("#tq-vehicle").value,
        pickup: document.querySelector("#tq-pickup").value,
        dropoff: document.querySelector("#tq-dropoff").value,
        hours: Number(document.querySelector("#tq-hours").value) || null,
      } });
      if (!next.data.ok) { out.textContent = next.data.error || "Could not get an estimate."; return; }
      const q = next.data.quote;
      if (!q) { out.textContent = "Choose a vehicle."; return; }
      if (!q.available) { out.textContent = "Price on request: " + q.message; return; }
      out.innerHTML = quoteBoxHtml({ total: (q.estimate ? "About " : "") + q.total, label: q.label + (q.miles ? ", " + q.miles + " mi" : ""), lines: q.lines, notes: q.notes });
    };
  };
  draw();
  renderAirportSettings(gen);
}

// Airport pickups on the booking page, passenger limits, and flight alerts (owner only).
async function renderAirportSettings(gen) {
  const result = await api("/api/booking-settings");
  if (gen !== renderGen) return;
  const box = document.querySelector("#airbox");
  if (!box || !result.data.ok) return;
  const s = result.data.settings;
  const airports = (s.airports && s.airports.airports) || [];
  const classes = result.data.fleetClasses || [];
  box.innerHTML = `<section class="card" id="airport-settings">
      <h2>Airport pickups</h2>
      <p class="sub">${airports.length ? "On the booking page: " + esc(airports.map((a) => a.code + " (" + a.spots.map((x) => x.label).join(", ") + ")").join("; ")) : "Off. Clients type both addresses."}</p>
      ${airports.length ? airports.map((a) => `<p class="sub">${a.pickupMeet ? "Pickup meeting points: " + esc(a.pickupMeet.options.map((o) => o.label).join(", ")) + ". " : ""}${a.dropoffStyle ? "Drop-off: " + esc(a.dropoffStyle.options.map((o) => o.label).join(" or ")) + "." : ""}</p>`).join("") : ""}
      <div class="actions"><button class="btn ghost" type="button" id="air-atl">Use the ATL terminals</button>${airports.length ? `<button class="btn ghost" type="button" id="air-off">Turn off airport options</button>` : ""}</div>
      <p class="sub">Your fleet: ${classes.length ? esc(classes.join(", ")) : "no vehicle types on the roster yet, so every vehicle on your rates is offered"}. Vehicles your fleet doesn't run are hidden from clients.</p>
      <label>Most passengers in a sedan<input id="cap-sedan" type="number" min="1" max="8" value="${esc(s.capacities && s.capacities.sedan ? s.capacities.sedan : 3)}"></label>
      <label>Alert drivers when a flight lands early by<select id="early-min">${[15, 20, 25, 30].map((n) => `<option value="${n}"${s.flightRules && s.flightRules.earlyMinutes === n ? " selected" : ""}>${n} minutes or more</option>`).join("")}</select></label>
      <p class="sub">Drivers are also alerted to delays over ${esc(s.flightRules ? s.flightRules.delayMinutes : 30)} minutes, cancellations, and other flight changes. ${result.data.flightTracking ? "Flight tracking is on." : "Flight tracking is not connected yet."}</p>
      <button class="btn" type="button" id="air-save">Save</button>
      <p class="sub" id="air-msg"></p>
    </section>`;
  const msg = document.querySelector("#air-msg");
  const save = async (body) => {
    const r = await api("/api/booking-settings", { method: "PUT", body });
    if (!r.data.ok) { msg.textContent = r.data.error || "Could not save."; return; }
    renderAirportSettings(gen);
  };
  document.querySelector("#air-atl").onclick = () => save({ preset: ["ATL"] });
  const off = document.querySelector("#air-off");
  if (off) off.onclick = () => save({ airports: { airports: [] } });
  document.querySelector("#air-save").onclick = () => save({
    capacities: { ...(s.capacities || {}), sedan: Number(document.querySelector("#cap-sedan").value) },
    flightRules: { ...(s.flightRules || {}), earlyMinutes: Number(document.querySelector("#early-min").value) },
  });
}

async function route() {
  locWatch.on = false;
  locWatch.rideId = "";
  locStopWatch();
  const gen = ++renderGen;
  const { name, id } = routeName();
  setTag();
  if (name === "where" && id) return renderWhere(id, gen);
  if (state.me && state.me.platformAdmin && name !== "admin" && name !== "login" && name !== "start") {
    location.hash = "#/admin";
    return;
  }
  if (name === "admin") {
    if (!state.me || !state.me.platformAdmin) {
      location.hash = state.me ? "#/rides" : "#/login";
      return;
    }
    return renderAdmin(gen);
  }
  if (billingLocked() && name !== "login" && name !== "start") return renderLocked();
  if (state.me && state.me.role === "driver" && name !== "rides" && name !== "login" && name !== "start") {
    location.hash = "#/rides";
    return;
  }
  if (!state.me && name !== "login" && name !== "start") return renderLogin();
  if (name === "login") return renderLogin();
  if (name === "start") return renderStart();
  if (name === "clients" && id === "new") return renderClientForm("", gen);
  if (name === "clients" && id) return renderClientForm(id, gen);
  if (name === "clients") return renderClients(gen);
  if (name === "new") return renderNewRide(gen);
  if (name === "rides" && id) return renderRide(id, gen);
  if (name === "invoices" && id) return renderInvoice(id, gen);
  if (name === "settings" && id === "rates") return renderRateSheet(gen);
  if (name === "settings") return renderSettings(gen);
  return renderRides(gen);
}

tabs.onclick = (event) => {
  const button = event.target.closest("[data-go]");
  if (button) location.hash = button.dataset.go;
};
window.addEventListener("hashchange", () => { route(); });
boot();
